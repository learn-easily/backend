`Character` → `abstract`: „უბრალოდ პერსონაჟს“ ვეღარ შექმნი, `Attack()` კი ყველა შვილს თავად უწერია. მკურნალობა — ინტერფეისი `IHealer` (ჯადოქარი და მღვდელი), ნადავლი — `ILootable`. საკლასოდან (`⭐`): `Paladin`, `Chest : ILootable`, მაღაზია ბრძოლამდე და წამალი ბრძოლაში — სრული თამაში: მაღაზია → ბრძოლა → ნადავლი.

უცვლელი ფაილები (`Warrior`, `Archer`, `Rogue`, `Berserker`) აქ აღარ მეორდება — იხ. ქვემოთ „პროექტი ამ ლექციის შემდეგ“.

**შემოწმება:** `new Character("???", 100)` → ❌ CS0144 · `Rogue` `Attack()`-ის გარეშე → ❌ CS0534 · ჯადოქარი ორ მასივშია (`party` და `healers`) · ნადავლი ერთი ციკლით გროვდება და ცოცხალ გმირებზე ნაწილდება.

```csharp Arena/Character.cs
namespace Arena
{
    internal abstract class Character
    {
        private string name = "";

        public string Name
        {
            get { return name; }
            set
            {
                if (value == "")
                {
                    name = "უცნობი";
                }
                else
                {
                    name = value;
                }
            }
        }

        public int Health { get; private set; }
        public int MaxHealth { get; private set; }
        public int Potions { get; private set; } = 3;
        public int Gold { get; private set; } = 50;

        public Character(string name, int maxHealth)
        {
            Name = name;
            if (maxHealth < 1) maxHealth = 1;
            MaxHealth = maxHealth;
            Health = maxHealth;
        }

        // შვილები ხედავენ, Main — ვერა
        protected int RollDamage(int min, int max)
        {
            return Random.Shared.Next(min, max + 1);
        }

        // ტანი აღარ აქვს — ყველა შვილი თავისას წერს
        public abstract int Attack();

        public virtual void TakeDamage(int damage)
        {
            if (damage < 0) return;

            Health -= damage;
            if (Health < 0) Health = 0;
        }

        public void Heal(int amount)
        {
            if (amount < 0 || !IsAlive()) return;

            Health += amount;
            if (Health > MaxHealth) Health = MaxHealth;
        }

        public bool UsePotion()
        {
            if (Potions == 0 || !IsAlive()) return false;

            Potions--;
            Heal(30);
            return true;
        }

        public bool IsAlive()
        {
            return Health > 0;
        }

        public void AddGold(int amount)
        {
            if (amount > 0) Gold += amount;
        }

        public bool BuyPotion()
        {
            if (Gold < 25) return false;

            Gold -= 25;
            Potions++;
            return true;
        }

        public override string ToString()
        {
            return $"{Name} ❤️ {Health}/{MaxHealth}";
        }
    }
}
```

```csharp Arena/Mage.cs
namespace Arena
{
    internal class Mage : Character, IHealer
    {
        public int Mana { get; private set; } = 60;

        public Mage(string name) : base(name, 80)
        {
        }

        public override int Attack()
        {
            if (Mana < 20) return RollDamage(3, 8);   // 🪄 ჯოხით — base.Attack() აღარ არსებობს

            Mana -= 20;
            return RollDamage(25, 35);
        }

        public void HealAlly(Character ally)
        {
            if (!IsAlive() || Mana < 10) return;

            Mana -= 10;
            ally.Heal(15);
        }

        public void FullRestore()
        {
            Heal(MaxHealth);
        }

        public override string ToString()
        {
            return base.ToString() + $" 🔮 {Mana}";
        }
    }
}
```

```csharp Arena/Chest.cs
namespace Arena
{
    // ⭐⭐ საკლასო: Chest-ს მშობელი არ ჰყავს — მხოლოდ ინტერფეისი. Open-ის ლოგიკა DropGold-ში გადავიდა
    internal class Chest : ILootable
    {
        private int gold;

        public Chest(int gold)
        {
            if (gold < 0) gold = 0;
            this.gold = gold;
        }

        public int DropGold()
        {
            int drop = gold;
            gold = 0;   // მეორედ — 0
            return drop;
        }
    }
}
```

```csharp Arena/Program.cs
namespace Arena
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Warrior luka = new Warrior("ლუკა");
            Mage ana = new Mage("ანა");
            Archer gio = new Archer("გიო");
            Priest mari = new Priest("მარი");
            Paladin beka = new Paladin("ბექა");   // ⭐ საკლასო
            Character[] party = { luka, ana, gio, mari, beka, new Rogue("დათო"), new Berserker("ნიკა") };
            IHealer[] healers = { ana, mari, beka };
            Dragon dragon = new Dragon();
            Chest chest = new Chest(100);

            // ⭐⭐ საკლასო: მაღაზია
            foreach (Character hero in party)
            {
                while (hero.BuyPotion()) { }
            }

            while (dragon.IsAlive() && AnyAlive(party))
            {
                foreach (Character hero in party)
                {
                    if (!hero.IsAlive() || !dragon.IsAlive()) continue;

                    if (hero.Health < 30) hero.UsePotion();   // ⭐⭐ საკლასო
                    dragon.TakeDamage(hero.Attack());
                }

                if (dragon.IsAlive())
                {
                    Character target = RandomAlive(party);
                    target.TakeDamage(dragon.Attack());
                    Console.WriteLine($"🔥 → {target}");
                }

                if (AnyAlive(party))
                {
                    Character weakest = FindWeakest(party);
                    foreach (IHealer healer in healers) healer.HealAlly(weakest);
                    Console.WriteLine($"✨ → {weakest}");
                }
                Thread.Sleep(600);
            }

            if (dragon.IsAlive())
            {
                Console.WriteLine("💀 დრაკონმა გაიმარჯვა...");
                return;
            }

            // ⭐⭐ საკლასო: Dragon და Chest — სხვადასხვა მშობელი, ერთი ინტერფეისი
            ILootable[] loot = { dragon, chest };
            int total = 0;
            foreach (ILootable l in loot) total += l.DropGold();

            int alive = 0;
            foreach (Character hero in party) if (hero.IsAlive()) alive++;

            foreach (Character hero in party)
            {
                if (hero.IsAlive()) hero.AddGold(total / alive);
            }
            Console.WriteLine($"🏆 გუნდმა გაიმარჯვა! 💰 {total} ოქრო — {alive} გმირს, თითოს {total / alive}");
        }

        static bool AnyAlive(Character[] team)
        {
            foreach (Character c in team)
            {
                if (c.IsAlive()) return true;
            }
            return false;
        }

        static Character RandomAlive(Character[] team)
        {
            Character target = team[Random.Shared.Next(team.Length)];
            while (!target.IsAlive())
            {
                target = team[Random.Shared.Next(team.Length)];
            }
            return target;
        }

        static Character FindWeakest(Character[] team)
        {
            Character weakest = RandomAlive(team);
            foreach (Character c in team)
            {
                if (c.IsAlive() && c.Health < weakest.Health) weakest = c;
            }
            return weakest;
        }
    }
}
```
