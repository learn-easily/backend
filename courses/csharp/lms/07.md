კონსოლის პროექტი **Arena** (.NET 10, Visual Studio, „Do not use top-level statements“). კლასში ვწერთ `Hero`-ს და ბრძოლის ციკლს უკვე ნასწავლით — კლასები და მეთოდები. `⭐ საკლასო` ნიშნით მონიშნულია საკლასო დავალების ამოხსნა (ხაფანგი, რაუნდის ბეჭდვა, გამარჯვებული) — შემდეგი ლექციები მასზე აგრძელებს.

**შემოწმება:** ბრძოლა რაუნდებად მიდის, ყოველ მე-3 რაუნდში რაინდი ხაფანგში ებმება, ბოლოს იბეჭდება გამარჯვებული. ლექციის ბოლოს — `knight.Health = 99999;` კომპილირდება: ეს შემდეგი ლექციის პრობლემაა.

```csharp Arena/Hero.cs
namespace Arena
{
    internal class Hero
    {
        public string Name { get; set; }
        public int Health { get; set; }

        public Hero(string name, int health)
        {
            Name = name;
            Health = health;
        }

        public int Attack()
        {
            return Random.Shared.Next(5, 21);
        }

        public void TakeDamage(int damage)
        {
            Health -= damage;
            if (Health < 0) Health = 0;
        }

        public bool IsAlive()
        {
            return Health > 0;
        }
    }
}
```
```csharp Arena/Program.cs
namespace Arena
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Hero knight = new Hero("რაინდი", 100);
            Hero dragon = new Hero("დრაკონი", 120);
            int round = 1;

            while (knight.IsAlive() && dragon.IsAlive())
            {
                PrintRound(round, knight, dragon);

                int damage = knight.Attack();
                dragon.TakeDamage(damage);
                Console.WriteLine($"⚔️ {knight.Name} → {damage}");

                if (dragon.IsAlive())
                {
                    damage = dragon.Attack();
                    knight.TakeDamage(damage);
                    Console.WriteLine($"🔥 {dragon.Name} → {damage}");
                }

                if (round % 3 == 0) Trap(knight);   // ⭐ საკლასო

                round++;
                Thread.Sleep(600);
            }

            Console.WriteLine($"🏆 {Winner(knight, dragon).Name} გაიმარჯვა!");
        }

        // ⭐ საკლასო: ობიექტი მეთოდს მისამართით გადაეცემა — Trap ნამდვილ რაინდს აზიანებს
        static void Trap(Hero hero)
        {
            hero.TakeDamage(15);
            Console.WriteLine($"🪤 {hero.Name} ხაფანგში გაება");
        }

        // ⭐⭐ საკლასო
        static void PrintRound(int round, Hero a, Hero b)
        {
            Console.WriteLine($"── რაუნდი {round} ── 🛡️ {a.Health} | 🐉 {b.Health}");
        }

        // ⭐⭐ საკლასო
        static Hero Winner(Hero a, Hero b)
        {
            if (a.IsAlive()) return a;
            return b;
        }
    }
}
```
