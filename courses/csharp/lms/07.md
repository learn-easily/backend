`Hero`-ს მონაცემი იმალება: `Name` წესით, `Health` / `MaxHealth` / `Potions` — `private set`, ცვლილება მხოლოდ მეთოდებით (`TakeDamage`, `Heal`, `UsePotion`). საკლასოდან (`⭐`): ოქრო, წამლის ყიდვა და `Chest` — ჯილდო ბრძოლის შემდეგ.

**შემოწმება:** `knight.Health = 99999;` → ❌ CS0272 · `knight.Name = ""` → „უცნობი“ · `TakeDamage(-50)` არაფერს ცვლის · ბრძოლამდე რაინდს 5 წამალი აქვს (3 + 2 ნაყიდი) · ზარდახშას მეორედ ვერ გახსნი.

```csharp Arena/Hero.cs
namespace Arena
{
    internal class Hero
    {
        private string name = "";

        public string Name
        {
            get { return name; }
            set
            {
                if (value == "")
                {
                    name = "უცნობი";
                }
                else
                {
                    name = value;
                }
            }
        }

        public int Health { get; private set; }
        public int MaxHealth { get; private set; }
        public int Potions { get; private set; } = 3;
        public int Gold { get; private set; } = 50;   // ⭐ საკლასო

        public Hero(string name, int maxHealth)
        {
            Name = name;
            if (maxHealth < 1) maxHealth = 1;
            MaxHealth = maxHealth;
            Health = maxHealth;
        }

        public int Attack()
        {
            return Random.Shared.Next(5, 21);
        }

        public void TakeDamage(int damage)
        {
            if (damage < 0) return;

            Health -= damage;
            if (Health < 0) Health = 0;
        }

        public void Heal(int amount)
        {
            if (amount < 0 || !IsAlive()) return;

            Health += amount;
            if (Health > MaxHealth) Health = MaxHealth;
        }

        public bool UsePotion()
        {
            if (Potions == 0 || !IsAlive()) return false;

            Potions--;
            Heal(30);
            return true;
        }

        public bool IsAlive()
        {
            return Health > 0;
        }

        // ⭐ საკლასო
        public void AddGold(int amount)
        {
            if (amount > 0) Gold += amount;
        }

        // ⭐⭐ საკლასო: 25 ოქრო → +1 წამალი
        public bool BuyPotion()
        {
            if (Gold < 25) return false;

            Gold -= 25;
            Potions++;
            return true;
        }
    }
}
```
```csharp Arena/Chest.cs
namespace Arena
{
    // ⭐⭐ საკლასო: ლექციის Chest (GetGold / AddGold) → ჯილდო ბრძოლის შემდეგ
    internal class Chest
    {
        private int gold;

        public bool IsOpen { get; private set; }

        public Chest(int gold)
        {
            if (gold < 0) gold = 0;
            this.gold = gold;
        }

        public void Open(Hero hero)
        {
            if (IsOpen || !hero.IsAlive()) return;

            IsOpen = true;
            hero.AddGold(gold);
            gold = 0;
        }
    }
}
```

```csharp Arena/Program.cs
namespace Arena
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Hero knight = new Hero("რაინდი", 100);
            Hero dragon = new Hero("დრაკონი", 150);
            Chest chest = new Chest(100);
            int round = 1;

            while (knight.BuyPotion()) { }   // ⭐⭐ საკლასო: მაღაზია ბრძოლამდე
            Console.WriteLine($"🧪 წამლები: {knight.Potions} · 💰 {knight.Gold}");

            while (knight.IsAlive() && dragon.IsAlive())
            {
                PrintRound(round, knight, dragon);

                int damage = knight.Attack();
                dragon.TakeDamage(damage);
                Console.WriteLine($"⚔️ {knight.Name} → {damage}");

                if (dragon.IsAlive())
                {
                    damage = dragon.Attack();
                    knight.TakeDamage(damage);
                    Console.WriteLine($"🔥 {dragon.Name} → {damage}");
                }

                if (round % 3 == 0) Trap(knight);

                if (knight.Health < 30 && knight.UsePotion())
                {
                    Console.WriteLine($"🧪 წამალი → ❤️ {knight.Health} (დარჩა {knight.Potions})");
                }

                round++;
                Thread.Sleep(600);
            }

            Console.WriteLine($"🏆 {Winner(knight, dragon).Name} გაიმარჯვა!");

            // ⭐⭐ საკლასო: ზარდახშა მხოლოდ ცოცხალ გმირს
            if (knight.IsAlive())
            {
                chest.Open(knight);
                Console.WriteLine($"💰 {knight.Name}-ს ოქრო: {knight.Gold}");
            }
        }

        static void Trap(Hero hero)
        {
            hero.TakeDamage(15);
            Console.WriteLine($"🪤 {hero.Name} ხაფანგში გაება");
        }

        static void PrintRound(int round, Hero a, Hero b)
        {
            Console.WriteLine($"── რაუნდი {round} ── 🛡️ {a.Health} | 🐉 {b.Health}");
        }

        static Hero Winner(Hero a, Hero b)
        {
            if (a.IsAlive()) return a;
            return b;
        }
    }
}
```
