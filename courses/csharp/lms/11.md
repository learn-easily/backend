OOP-ის ოთხივე პრინციპი ერთ პროექტში — ახალი კონსოლის პროექტი **Payments** (გადახდის სისტემა). ჯერ ვფიქრობთ, მერე ვწერთ: რა არსებებია (გადახდის სახეები), რა აქვთ საერთო (`Amount`, `Pay()`) → `abstract class Payment`; ვის შეუძლია თანხის დაბრუნება → `IRefundable`. ლექციის მეორე ნახევარში სტუდენტები იმავე ფორმით ზოოპარკს წერენ (საკლასო).

**შემოწმება:** ყველა გადახდა ერთი `Payment[]` ციკლით მუშაობს · საფულე ბალანსზე მეტს ვერ იხდის → `false` · `IRefundable[]`-ში ნაღდი ფული ვერ მოხვდება — მას დაბრუნება არ შეუძლია.

```csharp Payments/Payment.cs
namespace Payments
{
    internal abstract class Payment
    {
        public decimal Amount { get; private set; }

        public Payment(decimal amount)
        {
            if (amount < 0) amount = 0;
            Amount = amount;
        }

        public abstract bool Pay();

        public override string ToString()
        {
            return $"{Amount} ₾";
        }
    }
}
```

```csharp Payments/CashPayment.cs
namespace Payments
{
    internal class CashPayment : Payment

        }
```

```csharp Payments/CardPayment.cs
namespace Payments
{
    internal class CardPayment : Payment, IRefundable
    {
        private string cardNumber;

        public CardPayment(decimal amount, string number) : base(amount)
        {
            cardNumber = number;
        }

        public string Masked()
        {
            return "**** " + cardNumber.Substring(cardNumber.Length - 4);
        }

        public override bool Pay()
        {
            Console.WriteLine($"💳 ბარათი {Masked()} → {Amount} ₾");
            return true;
        }

        public void Refund()
        {
            Console.WriteLine($"↩️ {Amount} ₾ დაბრუნდა ბარათზე {Masked()}");
        }
    }
}
```
```csharp Payments/WalletPayment.cs
namespace Payments
{
    internal class WalletPayment : Payment, IRefundable
    {
        public decimal Balance { get; private set; }

        public WalletPayment(decimal amount, decimal balance) : base(amount)
        {
            Balance = balance;
        }

        public override bool Pay()
        {
            if (Balance < Amount)
            {
                Console.WriteLine($"❌ საფულეში მხოლოდ {Balance} ₾ — არ ჰყოფნის");
                return false;
            }

            Balance -= Amount;
            Console.WriteLine($"📱 საფულე → {Amount} ₾ (დარჩა {Balance} ₾)");
            return true;
        }

        public void Refund()
        {
            Balance += Amount;
            Console.WriteLine($"↩️ {Amount} ₾ დაბრუნდა საფულეში ({Balance} ₾)");
        }
    }
}
```
```csharp Payments/Program.cs
namespace Payments
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Payment[] payments =
            {
                new CardPayment(45, "4417123456789012"),
                new CashPayment(20),
                new WalletPayment(80, 50)
            };

            decimal total = 0;
            foreach (Payment p in payments)
            {
                if (p.Pay()) total += p.Amount;
            }
            Console.WriteLine($"💰 სულ გადახდილია: {total} ₾");

            CardPayment card = new CardPayment(30, "5555666677778888");
            WalletPayment wallet = new WalletPayment(10, 100);
            card.Pay();
            wallet.Pay();

            IRefundable[] refunds = { card, wallet };
            foreach (IRefundable r in refunds)
            {
                r.Refund();
            }
        }
    }
}
```
