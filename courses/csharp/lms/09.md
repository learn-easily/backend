პოლიმორფიზმი: `virtual` / `override` — `Smash`, `Fireball`, `FireBreath`, `Rage` ქრება, ყველა თავის `Attack()`-ს წერს. გუნდი ერთ `Character[]`-შია, ბრძოლა — ერთი ციკლი. ახალი `Archer`, მეომარს ჯავშანი (`override TakeDamage`), `ToString()`. საკლასოდან (`⭐`): `Rogue`, ბერსერკი პოლიმორფულად, ზარდახშას შემთხვევითი ცოცხალი გმირი ხსნის.

**შემოწმება:** ორი ახალი გმირი დაემატა — ბრძოლის ციკლი არ შეცვლილა · `Console.WriteLine(berserker)` → „🪓 ნიკა ❤️ 100/100“ · `virtual`-ის გარეშე `override` → ❌ CS0506.

```csharp Arena/Character.cs
namespace Arena
{
    internal class Character
    {
        private string name = "";

        public string Name
        {
            get { return name; }
            set
            {
                if (value == "")
                {
                    name = "უცნობი";
                }
                else
                {
                    name = value;
                }
            }
        }

        public int Health { get; private set; }
        public int MaxHealth { get; private set; }
        public int Potions { get; private set; } = 3;
        public int Gold { get; private set; } = 50;

        public Character(string name, int maxHealth)
        {
            Name = name;
            if (maxHealth < 1) maxHealth = 1;
            MaxHealth = maxHealth;
            Health = maxHealth;
        }

        // შვილები ხედავენ, Main — ვერა
        protected int RollDamage(int min, int max)
        {
            return Random.Shared.Next(min, max + 1);
        }

        public virtual int Attack()
        {
            return RollDamage(5, 15);
        }

        public virtual void TakeDamage(int damage)
        {
            if (damage < 0) return;

            Health -= damage;
            if (Health < 0) Health = 0;
        }

        public void Heal(int amount)
        {
            if (amount < 0 || !IsAlive()) return;

            Health += amount;
            if (Health > MaxHealth) Health = MaxHealth;
        }

        public bool UsePotion()
        {
            if (Potions == 0 || !IsAlive()) return false;

            Potions--;
            Heal(30);
            return true;
        }

        public bool IsAlive()
        {
            return Health > 0;
        }

        public void AddGold(int amount)
        {
            if (amount > 0) Gold += amount;
        }

        public bool BuyPotion()
        {
            if (Gold < 25) return false;

            Gold -= 25;
            Potions++;
            return true;
        }

        public override string ToString()
        {
            return $"{Name} ❤️ {Health}/{MaxHealth}";
        }
    }
}
```

```csharp Arena/Warrior.cs
namespace Arena
{
    internal class Warrior : Character
    {
        public int Armor { get; private set; } = 5;

        public Warrior(string name) : base(name, 120)
        {
        }

        public override int Attack()   // Smash()-ის ნაცვლად
        {
            return RollDamage(15, 25);
        }

        public override void TakeDamage(int damage)
        {
            int reduced = damage - Armor;
            if (reduced < 0) reduced = 0;
            base.TakeDamage(reduced);
        }
    }
}
```

```csharp Arena/Mage.cs
namespace Arena
{
    internal class Mage : Character
    {
        public int Mana { get; private set; } = 60;

        public Mage(string name) : base(name, 80)
        {
        }

        public override int Attack()   // Fireball()-ის ნაცვლად
        {
            if (Mana < 20) return base.Attack();

            Mana -= 20;
            return RollDamage(25, 35);
        }

        public void FullRestore()
        {
            Heal(MaxHealth);
        }

        public override string ToString()
        {
            return base.ToString() + $" 🔮 {Mana}";
        }
    }
}
```

```csharp Arena/Rogue.cs
    {
        public Rogue(string name) : base(name, 85)
        {
        }

        public override int Attack()
        {
            int damage = RollDamage(6, 12);
            if (Random.Shared.Next(2) == 0)
            {
                damage += RollDamage(6, 12);
            }
            return damage;
        }
    }
```

```csharp Arena/Program.cs
namespace Arena
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Character[] party =
            {
                new Warrior("ლუკა"),
                new Mage("ანა"),
                new Archer("გიო"),
                new Rogue("დათო"),        // ⭐ საკლასო
                new Berserker("ნიკა")     // ⭐⭐ საკლასო
            };
            Dragon dragon = new Dragon();
            Chest chest = new Chest(100);

            while (dragon.IsAlive() && AnyAlive(party))
            {
                Console.WriteLine($"── {dragon}");

                foreach (Character hero in party)
                {
                    if (hero.IsAlive() && dragon.IsAlive())
                    {
                        dragon.TakeDamage(hero.Attack());
                    }
                }

                if (dragon.IsAlive())
                {
                    Character target = RandomAlive(party);
                    target.TakeDamage(dragon.Attack());
                    Console.WriteLine($"🔥 → {target}");
                }
                Thread.Sleep(600);
            }

            if (dragon.IsAlive())
            {
                Console.WriteLine("💀 დრაკონმა გაიმარჯვა...");
                return;
            }

            Console.WriteLine("🏆 გუნდმა გაიმარჯვა!");

            // ⭐⭐ საკლასო: იღბლიანი გმირი
            Character lucky = RandomAlive(party);
            chest.Open(lucky);
            Console.WriteLine($"💰 ზარდახშა გახსნა: {lucky} · ოქრო {lucky.Gold}");
        }

        static bool AnyAlive(Character[] team)
        {
            foreach (Character c in team)
            {
                if (c.IsAlive()) return true;
            }
            return false;
        }

        static Character RandomAlive(Character[] team)
        {
            Character target = team[Random.Shared.Next(team.Length)];
            while (!target.IsAlive())
            {
                target = team[Random.Shared.Next(team.Length)];
            }
            return target;
        }
    }
}
```
