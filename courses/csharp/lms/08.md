`Hero` → `Character` (კლასის სახელზე **F2** — Rename, VS ყველგან შეცვლის): საერთო კოდი მშობელშია, `Warrior`, `Mage`, `Dragon` — შვილები `: base(...)`-ით. `RollDamage` — `protected`. საკლასოდან (`⭐`): `Berserker` და ზარდახშა გუნდისთვის (`Open(Character)`).

**შემოწმება:** `luka.TakeDamage(30)` მშობლისგან მუშაობს · `warrior.RollDamage(1, 2)` Main-ში → ❌ CS0122 · ბერსერკი ყოველ `Rage()`-ზე 5-ით სუსტდება · ბრძოლაში დრაკონი ჯერ მეომარს ურტყამს, მერე ბერსერკს, მერე ჯადოქარს.

```csharp Arena/Character.cs
namespace Arena
{
    internal class Character
    {
        public string Name { get; private set; }
        public int Health { get; private set; }
        public int MaxHealth { get; private set; }
        public int Potions { get; private set; } = 3;
        public int Gold { get; private set; } = 50;

        public Character(string name, int maxHealth)
        {
            if (name == "") name = "უცნობი";
            Name = name;
            if (maxHealth < 1) maxHealth = 1;
            MaxHealth = maxHealth;
            Health = maxHealth;
        }

        // შვილები ხედავენ, Main — ვერა
        protected int RollDamage(int min, int max)
        {
            return Random.Shared.Next(min, max + 1);
        }

        public int Attack()
        {
            return RollDamage(5, 15);
        }

        public void TakeDamage(int damage)
        {
            if (damage < 0) return;

            Health -= damage;
            if (Health < 0) Health = 0;
        }

        public void Heal(int amount)
        {
            if (amount < 0 || !IsAlive()) return;

            Health += amount;
            if (Health > MaxHealth) Health = MaxHealth;
        }

        public bool UsePotion()
        {
            if (Potions == 0 || !IsAlive()) return false;

            Potions--;
            Heal(30);
            return true;
        }

        public bool IsAlive()
        {
            return Health > 0;
        }

        public void AddGold(int amount)
        {
            if (amount > 0) Gold += amount;
        }

        public bool BuyPotion()
        {
            if (Gold < 25) return false;

            Gold -= 25;
            Potions++;
            return true;
        }
    }
}
```

```csharp Arena/Mage.cs
namespace Arena
{
    internal class Mage : Character
    {
        public int Mana { get; private set; } = 60;

        public Mage(string name) : base(name, 80)
        {
        }

        public int Fireball()
        {
            if (Mana < 20) return Attack();

            Mana -= 20;
            return RollDamage(25, 35);
        }

        public void FullRestore()
        {
            Heal(MaxHealth);   // Health = MaxHealth აქ ვერ დაიწერება — private set
        }
    }
}
```

```csharp Arena/Chest.cs
namespace Arena
{
    // ზარდახშა პერსონაჟი არ არის — ამიტომ Chest : Character არ იქნება
    internal class Chest
    {
        private int gold;

        public bool IsOpen { get; private set; }

        public Chest(int gold)
        {
            if (gold < 0) gold = 0;
            this.gold = gold;
        }

        public void Open(Character hero)   // ⭐⭐ საკლასო: Hero → Character
        {
            if (IsOpen || !hero.IsAlive()) return;

            IsOpen = true;
            hero.AddGold(gold);
            gold = 0;
        }
    }
}
```

```csharp Arena/Program.cs
namespace Arena
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            Warrior warrior = new Warrior("ლუკა");
            Mage mage = new Mage("ანა");
            Berserker berserker = new Berserker("ნიკა");   // ⭐⭐ საკლასო
            Dragon dragon = new Dragon();
            Chest chest = new Chest(100);
            int round = 1;

            while (dragon.IsAlive() && (warrior.IsAlive() || berserker.IsAlive() || mage.IsAlive()))
            {
                Console.WriteLine($"── {round} ── 🛡️ {warrior.Health} | 🪓 {berserker.Health} | 🔮 {mage.Health} | 🐉 {dragon.Health}");

                if (warrior.IsAlive()) dragon.TakeDamage(warrior.Smash());
                if (berserker.IsAlive()) dragon.TakeDamage(berserker.Rage());
                if (mage.IsAlive()) dragon.TakeDamage(mage.Fireball());

                if (dragon.IsAlive())
                {
                    int damage = dragon.FireBreath();
                    if (warrior.IsAlive()) warrior.TakeDamage(damage);
                    else if (berserker.IsAlive()) berserker.TakeDamage(damage);
                    else mage.TakeDamage(damage);
                }

                round++;
                Thread.Sleep(600);
            }

            if (dragon.IsAlive())
            {
                Console.WriteLine("💀 დრაკონმა გაიმარჯვა...");
                return;
            }

            Console.WriteLine("🏆 გუნდმა გაიმარჯვა!");

            // ⭐⭐ საკლასო: ზარდახშას პირველი ცოცხალი გმირი ხსნის
            Character opener = mage;
            if (warrior.IsAlive()) opener = warrior;
            else if (berserker.IsAlive()) opener = berserker;
            chest.Open(opener);
            Console.WriteLine($"💰 {opener.Name}: {opener.Gold} ოქრო");
        }
    }
}
```
