```csharp Common/Result.cs
namespace Lms.Api.Common;

public record Error(string Code, string Message)
{
    public static readonly Error None = new("", "");
}

public class Result
{
    public bool IsSuccess { get; }
    public bool IsFailure => !IsSuccess;
    public Error Error { get; }

    protected Result(bool isSuccess, Error error)
    {
        IsSuccess = isSuccess;
        Error = error;
    }

    public static Result Success() => new(true, Error.None);
    public static Result Failure(Error error) => new(false, error);
}

public class Result<T> : Result
{
    private readonly T? _value;

    private Result(T? value, bool isSuccess, Error error) : base(isSuccess, error)
    {
        _value = value;
    }

    public T Value => IsSuccess
        ? _value!
        : throw new InvalidOperationException("შეცდომას მნიშვნელობა არ აქვს");

    public static Result<T> Success(T value) => new(value, true, Error.None);
    public static new Result<T> Failure(Error error) => new(default, false, error);
}
```

```csharp Entities/Course.cs
using Lms.Api.Common;

namespace Lms.Api.Entities;

public class Course
{
    public const int MinWeeks = 1;
    public const int MaxWeeks = 52;

    public Guid Id { get; private set; }
    public string Title { get; private set; } = string.Empty;
    public string? Description { get; private set; }
    public int DurationWeeks { get; private set; }
    public DateTime CreatedAt { get; private set; }

    private Course() { }   // EF Core-ისთვის

    public static Result<Course> Create(string title, string? description, int durationWeeks)
    {
        var check = Validate(title, durationWeeks);
        if (check.IsFailure)
            return Result<Course>.Failure(check.Error);

        var course = new Course
        {
            Id = Guid.CreateVersion7(),
            Title = title.Trim(),
            Description = description?.Trim(),
            DurationWeeks = durationWeeks,
            CreatedAt = DateTime.UtcNow
        };
        return Result<Course>.Success(course);
    }

    public Result Update(string title, string? description, int durationWeeks)
    {
        var check = Validate(title, durationWeeks);
        if (check.IsFailure)
            return check;

        Title = title.Trim();
        Description = description?.Trim();
        DurationWeeks = durationWeeks;
        return Result.Success();
    }

    private static Result Validate(string title, int durationWeeks)
    {
        if (string.IsNullOrWhiteSpace(title))
            return Result.Failure(CourseErrors.TitleRequired);
        if (durationWeeks is < MinWeeks or > MaxWeeks)
            return Result.Failure(CourseErrors.InvalidDuration);
        return Result.Success();
    }
}
```

```csharp Entities/Email.cs
using System.Net.Mail;
using Lms.Api.Common;

namespace Lms.Api.Entities;

public sealed record Email
{
    public const int MaxLength = 200;

    public string Value { get; }

    private Email(string value)
    {
        Value = value;
    }

    public static Result<Email> Create(string? input)
    {
        if (string.IsNullOrWhiteSpace(input))
            return Result<Email>.Failure(EmailErrors.Required);

        var value = input.Trim().ToLowerInvariant();
        if (value.Length > MaxLength ||
            !MailAddress.TryCreate(value, out var address) ||
            address.Address != value)
            return Result<Email>.Failure(EmailErrors.Invalid);

        return Result<Email>.Success(new Email(value));
    }

    // ბაზიდან წაკითხვისთვის: იქ მხოლოდ შემოწმებული Email ხვდება
    public static Email FromDatabase(string value) => new(value);

    public override string ToString() => Value;
}
```

```csharp Data/DataContext.cs
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Data;

public class DataContext : DbContext
{
    public DataContext(DbContextOptions<DataContext> options)
        : base(options) { }

    public DbSet<Course> Courses { get; set; }
    public DbSet<Student> Students { get; set; }
    public DbSet<Lecturer> Lecturers { get; set; }
    public DbSet<Group> Groups { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Course>(course =>
        {
            course.Property(c => c.Title).HasMaxLength(100);
            course.HasIndex(c => c.Title).IsUnique();
        });

        modelBuilder.Entity<Student>(student =>
        {
            student.Property(x => x.FirstName).HasMaxLength(50);
            student.Property(x => x.LastName).HasMaxLength(50);
            student.Property(x => x.Email)
                .HasConversion(
                    email => email.Value,
                    value => Email.FromDatabase(value))
                .HasMaxLength(Email.MaxLength);
            student.HasIndex(x => x.Email).IsUnique();
        });

        modelBuilder.Entity<Lecturer>(lecturer =>
        {
            lecturer.Property(x => x.FirstName).HasMaxLength(50);
            lecturer.Property(x => x.LastName).HasMaxLength(50);
            lecturer.Property(x => x.Email)
                .HasConversion(
                    email => email.Value,
                    value => Email.FromDatabase(value))
                .HasMaxLength(Email.MaxLength);
            lecturer.HasIndex(x => x.Email).IsUnique();
        });

        modelBuilder.Entity<Group>(group =>
        {
            group.Property(g => g.Name).HasMaxLength(50);
        });
    }
}
```

```csharp Features/Courses/CreateCourse/CreateCourseHandler.cs
using Lms.Api.Data;
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Features.Courses.CreateCourse;

public class CreateCourseHandler
{
    private readonly DataContext _db;

    public CreateCourseHandler(DataContext db)
    {
        _db = db;
    }

    public async Task<IResult> HandleAsync(CreateCourseRequest request)
    {
        var result = Course.Create(request.Title, request.Description, request.DurationWeeks);
        if (result.IsFailure)
            return Results.BadRequest(result.Error);

        var course = result.Value;
        if (await _db.Courses.AnyAsync(c => c.Title == course.Title))
            return Results.Conflict(CourseErrors.TitleTaken);

        _db.Courses.Add(course);
        await _db.SaveChangesAsync();
        return Results.Created($"/api/courses/{course.Id}", course.ToResponse());
    }
}
```
