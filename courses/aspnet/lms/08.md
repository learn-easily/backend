```csharp Program.cs
    app.UseSwagger();
    app.UseSwaggerUI();
```

```csharp Data/DataContext.cs
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Data;

public class DataContext : DbContext
{
    public DataContext(DbContextOptions<DataContext> options)
        : base(options) { }

    public DbSet<Course> Courses { get; set; }
    public DbSet<Student> Students { get; set; }
    public DbSet<Lecturer> Lecturers { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Course>(course =>
        {
            course.Property(c => c.Title).HasMaxLength(100);
            course.HasIndex(c => c.Title).IsUnique();
        });

        modelBuilder.Entity<Student>(student =>
        {
            student.Property(s => s.FirstName).HasMaxLength(50);
            student.Property(s => s.LastName).HasMaxLength(50);
            student.Property(s => s.Email).HasMaxLength(200);
            student.HasIndex(s => s.Email).IsUnique();
        });

        modelBuilder.Entity<Lecturer>(lecturer =>
        {
            lecturer.Property(l => l.FirstName).HasMaxLength(50);
            lecturer.Property(l => l.LastName).HasMaxLength(50);
            lecturer.Property(l => l.Email).HasMaxLength(200);
            lecturer.HasIndex(l => l.Email).IsUnique();
        });
    }
}
```

```csharp Mappers/CourseMapper.cs
using Lms.Api.Dtos;
using Lms.Api.Entities;

namespace Lms.Api.Mappers;

public static class CourseMapper
{
    public static CourseResponse ToResponse(this Course course)
    {
        return new CourseResponse(course.Id, course.Title,
            course.Description, course.DurationWeeks, course.CreatedAt);
    }

    public static Course ToEntity(this CreateCourseRequest request)
    {
        return new Course
        {
            Title = request.Title.Trim(),
            Description = request.Description,
            DurationWeeks = request.DurationWeeks,
            CreatedAt = DateTime.UtcNow
        };
    }

    public static void UpdateFrom(this Course course, UpdateCourseRequest request)
    {
        course.Title = request.Title.Trim();
        course.Description = request.Description;
        course.DurationWeeks = request.DurationWeeks;
    }
}
```

```csharp Controllers/CoursesController.cs
using Lms.Api.Data;
using Lms.Api.Dtos;
using Lms.Api.Mappers;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Controllers;

[ApiController]
[Route("api/courses")]
public class CoursesController : ControllerBase
{
    private readonly DataContext _db;

    public CoursesController(DataContext db)
    {
        _db = db;
    }

    [HttpGet]
    public async Task<IActionResult> GetAll()
    {
        var courses = await _db.Courses
            .OrderBy(c => c.Title)
            .ToListAsync();
        return Ok(courses.Select(c => c.ToResponse()));
    }

    [HttpGet("{id:guid}")]
    public async Task<IActionResult> GetById(Guid id)
    {
        var course = await _db.Courses.FindAsync(id);
        if (course is null) return NotFound();
        return Ok(course.ToResponse());
    }

    [HttpPost]
    public async Task<IActionResult> Create(CreateCourseRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.Title))
            return BadRequest("კურსის სახელი სავალდებულოა");

        if (request.DurationWeeks is < 1 or > 52)
            return BadRequest("ხანგრძლივობა 1-დან 52 კვირამდეა");

        var title = request.Title.Trim();
        if (await _db.Courses.AnyAsync(c => c.Title == title))
            return Conflict($"კურსი „{title}“ უკვე არსებობს");

        var course = request.ToEntity();
        _db.Courses.Add(course);
        await _db.SaveChangesAsync();
        return CreatedAtAction(nameof(GetById),
            new { id = course.Id }, course.ToResponse());
    }

    [HttpPut("{id:guid}")]
    public async Task<IActionResult> Update(Guid id, UpdateCourseRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.Title))
            return BadRequest("კურსის სახელი სავალდებულოა");
        if (request.DurationWeeks is < 1 or > 52)
            return BadRequest("ხანგრძლივობა 1-დან 52 კვირამდეა");

        var course = await _db.Courses.FindAsync(id);
        if (course is null) return NotFound();

        var title = request.Title.Trim();
        if (await _db.Courses.AnyAsync(c => c.Title == title && c.Id != id))
            return Conflict($"კურსი „{title}“ უკვე არსებობს");

        course.UpdateFrom(request);
        await _db.SaveChangesAsync();
        return NoContent();
    }

    [HttpDelete("{id:guid}")]
    public async Task<IActionResult> Delete(Guid id)
    {
        var course = await _db.Courses.FindAsync(id);
        if (course is null) return NotFound();
        _db.Courses.Remove(course);
        await _db.SaveChangesAsync();
        return NoContent();
    }
}
```

```csharp Controllers/StudentsController.cs
using Lms.Api.Data;
using Lms.Api.Dtos;
using Lms.Api.Mappers;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Controllers;

[ApiController]
[Route("api/students")]
public class StudentsController : ControllerBase
{
    private readonly DataContext _db;

    public StudentsController(DataContext db)
    {
        _db = db;
    }

    [HttpGet]
    public async Task<IActionResult> GetAll()
    {
        var items = await _db.Students
            .OrderBy(x => x.LastName).ThenBy(x => x.FirstName)
            .ToListAsync();
        return Ok(items.Select(x => x.ToResponse()));
    }

    [HttpGet("{id:guid}")]
    public async Task<IActionResult> GetById(Guid id)
    {
        var student = await _db.Students.FindAsync(id);
        if (student is null) return NotFound();
        return Ok(student.ToResponse());
    }

    [HttpPost]
    public async Task<IActionResult> Create(CreateStudentRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.FirstName) ||
            string.IsNullOrWhiteSpace(request.LastName) ||
            string.IsNullOrWhiteSpace(request.Email))
            return BadRequest("სახელი, გვარი და Email სავალდებულოა");

        var email = request.Email.Trim().ToLowerInvariant();
        if (await _db.Students.AnyAsync(x => x.Email == email))
            return Conflict("ამ Email-ით სტუდენტი უკვე არსებობს");

        var student = request.ToEntity(email);
        _db.Students.Add(student);
        await _db.SaveChangesAsync();
        return CreatedAtAction(nameof(GetById), new { id = student.Id }, student.ToResponse());
    }

    [HttpPut("{id:guid}")]
    public async Task<IActionResult> Update(Guid id, UpdateStudentRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.FirstName) ||
            string.IsNullOrWhiteSpace(request.LastName) ||
            string.IsNullOrWhiteSpace(request.Email))
            return BadRequest("სახელი, გვარი და Email სავალდებულოა");

        var student = await _db.Students.FindAsync(id);
        if (student is null) return NotFound();

        var email = request.Email.Trim().ToLowerInvariant();
        if (await _db.Students.AnyAsync(x => x.Email == email && x.Id != id))
            return Conflict("ამ Email-ით სტუდენტი უკვე არსებობს");

        student.UpdateFrom(request, email);
        await _db.SaveChangesAsync();
        return NoContent();
    }

    [HttpDelete("{id:guid}")]
    public async Task<IActionResult> Delete(Guid id)
    {
        var student = await _db.Students.FindAsync(id);
        if (student is null) return NotFound();
        _db.Students.Remove(student);
        await _db.SaveChangesAsync();
        return NoContent();
    }
}
```
