ყველა feature **Vertical Slices**-ზე: თითო ოპერაცია = Request · Handler · Endpoint. Controllers, `Dtos/` და `Mappers/` იშლება; საერთო რჩება `Entities/`, `Data/` და თითო feature-ის `…Response` + `…Mapper` (`ToResponse`).

Package-ები: `Swashbuckle.AspNetCore` → წაშალე · `Scalar.AspNetCore` → დაამატე. Swagger-ის ნაცვლად: `/scalar`.

using Lms.Api.Data;
using Lms.Api.Features.Courses;
using Lms.Api.Features.Lecturers;
using Lms.Api.Features.Students;
using Microsoft.EntityFrameworkCore;
using Scalar.AspNetCore;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddOpenApi();
builder.Services.AddDbContext<DataContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("Default")));

builder.Services.AddCourseHandlers();
builder.Services.AddStudentHandlers();
builder.Services.AddLecturerHandlers();

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.MapOpenApi();
    app.MapScalarApiReference();
}

app.UseHttpsRedirection();

app.MapCourseEndpoints();
app.MapStudentEndpoints();
app.MapLecturerEndpoints();

app.Run();

using Lms.Api.Data;
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Features.Courses.CreateCourse;

public class CreateCourseHandler
{
    private readonly DataContext _db;

    public CreateCourseHandler(DataContext db)
    {
        _db = db;
    }

    public async Task<IResult> HandleAsync(CreateCourseRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.Title))
            return Results.BadRequest("კურსის სახელი სავალდებულოა");
        if (request.DurationWeeks is < 1 or > 52)
            return Results.BadRequest("ხანგრძლივობა 1-დან 52 კვირამდეა");

        var title = request.Title.Trim();
        if (await _db.Courses.AnyAsync(c => c.Title == title))
            return Results.Conflict($"კურსი „{title}“ უკვე არსებობს");

        var course = new Course
        {
            Title = title,
            Description = request.Description,
            DurationWeeks = request.DurationWeeks,
            CreatedAt = DateTime.UtcNow
        };
        _db.Courses.Add(course);
        await _db.SaveChangesAsync();
        return Results.Created($"/api/courses/{course.Id}", course.ToResponse());
    }
}

using Lms.Api.Features.Lecturers.GetLecturers;
using Lms.Api.Features.Lecturers.GetLecturerById;
using Lms.Api.Features.Lecturers.UpdateLecturer;
using Lms.Api.Features.Lecturers.DeleteLecturer;

namespace Lms.Api.Features.Lecturers;

public static class LecturerEndpoints
{
    public static void AddLecturerHandlers(this IServiceCollection services)
    {
        services.AddScoped<CreateLecturerHandler>();
        services.AddScoped<GetLecturersHandler>();
        services.AddScoped<GetLecturerByIdHandler>();
        services.AddScoped<UpdateLecturerHandler>();
        services.AddScoped<DeleteLecturerHandler>();
    }

    public static void MapLecturerEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/lecturers").WithTags("Lecturers");

        CreateLecturerEndpoint.Map(group);
        GetLecturersEndpoint.Map(group);
        GetLecturerByIdEndpoint.Map(group);
        UpdateLecturerEndpoint.Map(group);
        DeleteLecturerEndpoint.Map(group);
    }
}
