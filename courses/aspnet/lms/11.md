**Aggregate:** `Group` (root) + `Enrollment`-ები. ჩარიცხვა მხოლოდ `group.Enroll(...)`-ით, `DbSet<Enrollment>` არ არის. **Concurrency:** `RowVersion` + `EnrolledCount++` (root-ის რიგი ყოველ ჩარიცხვაზე იცვლება). ჯგუფი ↔ კურსი / ლექტორი — **Restrict**, ჯგუფი → ჩარიცხვები — **Cascade**.

```
Add-Migration AddEnrollments
Update-Database
```

**შემოწმება:** ჯგუფი `capacity: 2` → სამი სხვადასხვა სტუდენტი → მესამე **409** `Group.Full` · იგივე სტუდენტი მეორედ → **409** `Group.AlreadyEnrolled` · ჯგუფიანი კურსის წაშლა → **409** `Course.HasGroups`.

**race condition-ის ცდა:** `EnrollStudentHandler`-ში დროებით `await Task.Delay(5000);` `Enroll`-სა და `SaveChangesAsync`-ს შორის → Scalar-ის ორი ჩანართიდან ბოლო ადგილზე ერთდროულად → ერთი **200**, მეორე **409** `Group.ConcurrentChange`. ცდის მერე `Task.Delay` წაშალე.

```csharp Entities/Group.cs
using Lms.Api.Common;

namespace Lms.Api.Entities;

public class Group
{
    public const int MaxCapacity = 20;

    private readonly List<Enrollment> _enrollments = [];

    public Guid Id { get; private set; }
    public Guid CourseId { get; private set; }
    public Guid LecturerId { get; private set; }
    public string Name { get; private set; } = string.Empty;
    public DateOnly StartDate { get; private set; }
    public int Capacity { get; private set; }
    public int EnrolledCount { get; private set; }
    public byte[] RowVersion { get; private set; } = [];

    public IReadOnlyCollection<Enrollment> Enrollments => _enrollments;

    private Group() { }   // EF Core-ისთვის

    public static Result<Group> Create(Guid courseId, Guid lecturerId, string name,
        DateOnly startDate, int capacity, DateOnly today)
    {
        if (string.IsNullOrWhiteSpace(name))
            return Result<Group>.Failure(GroupErrors.NameRequired);
        if (capacity is < 1 or > MaxCapacity)
            return Result<Group>.Failure(GroupErrors.InvalidCapacity);
        if (startDate < today)
            return Result<Group>.Failure(GroupErrors.StartInPast);

        return Result<Group>.Success(new Group
        {
            Id = Guid.CreateVersion7(),
            CourseId = courseId,
            LecturerId = lecturerId,
            Name = name.Trim(),
            StartDate = startDate,
            Capacity = capacity
        });
    }

    // R3: ადგილი · ერთხელ · დაწყებულ ჯგუფში მხოლოდ პირველ კვირაში
    public Result Enroll(Guid studentId, DateOnly today, DateTime now)
    {
        if (_enrollments.Any(e => e.StudentId == studentId))
            return Result.Failure(GroupErrors.AlreadyEnrolled);
        if (EnrolledCount >= Capacity)
            return Result.Failure(GroupErrors.Full);
        if (today >= StartDate.AddDays(7))
            return Result.Failure(GroupErrors.EnrollmentClosed);

        _enrollments.Add(new Enrollment(Id, studentId, now));
        EnrolledCount++;   // root-ის რიგი იცვლება → RowVersion იზრდება
        return Result.Success();
    }
}
```

```csharp Data/DataContext.cs
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Data;

public class DataContext : DbContext
{
    public DataContext(DbContextOptions<DataContext> options)
        : base(options) { }

    public DbSet<Course> Courses { get; set; }
    public DbSet<Student> Students { get; set; }
    public DbSet<Lecturer> Lecturers { get; set; }
    public DbSet<Group> Groups { get; set; }
    // DbSet<Enrollment> არ არის: ჩარიცხვა მხოლოდ group.Enroll-ით

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Course>(course =>
        {
            course.Property(c => c.Title).HasMaxLength(100);
            course.HasIndex(c => c.Title).IsUnique();
        });

        modelBuilder.Entity<Student>(student =>
        {
            student.Property(x => x.FirstName).HasMaxLength(50);
            student.Property(x => x.LastName).HasMaxLength(50);
            student.Property(x => x.Email)
                .HasConversion(
                    email => email.Value,
                    value => Email.FromDatabase(value))
                .HasMaxLength(Email.MaxLength);
            student.HasIndex(x => x.Email).IsUnique();
        });

        modelBuilder.Entity<Lecturer>(lecturer =>
        {
            lecturer.Property(x => x.FirstName).HasMaxLength(50);
            lecturer.Property(x => x.LastName).HasMaxLength(50);
            lecturer.Property(x => x.Email)
                .HasConversion(
                    email => email.Value,
                    value => Email.FromDatabase(value))
                .HasMaxLength(Email.MaxLength);
            lecturer.HasIndex(x => x.Email).IsUnique();
        });

        modelBuilder.Entity<Group>(group =>
        {
            group.Property(g => g.Name).HasMaxLength(50);
            group.Property(g => g.RowVersion).IsRowVersion();

            group.HasOne<Course>().WithMany()
                .HasForeignKey(g => g.CourseId).OnDelete(DeleteBehavior.Restrict);
            group.HasOne<Lecturer>().WithMany()
                .HasForeignKey(g => g.LecturerId).OnDelete(DeleteBehavior.Restrict);

            group.HasMany(g => g.Enrollments)
                .WithOne()
                .HasForeignKey(e => e.GroupId)
                .OnDelete(DeleteBehavior.Cascade);
        });

        modelBuilder.Entity<Enrollment>(enrollment =>
        {
            enrollment.HasOne<Student>().WithMany()
                .HasForeignKey(e => e.StudentId)
                .OnDelete(DeleteBehavior.Restrict);
            enrollment.HasIndex(e => new { e.GroupId, e.StudentId }).IsUnique();
        });
    }
}
```

```csharp Features/Courses/DeleteCourse/DeleteCourseHandler.cs
using Lms.Api.Data;
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Features.Courses.DeleteCourse;

public class DeleteCourseHandler
{
    private readonly DataContext _db;

    public DeleteCourseHandler(DataContext db)
    {
        _db = db;
    }

    public async Task<IResult> HandleAsync(Guid id)
    {
        var course = await _db.Courses.FindAsync(id);
        if (course is null)
            return Results.NotFound();

        if (await _db.Groups.AnyAsync(g => g.CourseId == id))
            return Results.Conflict(CourseErrors.HasGroups);

        _db.Courses.Remove(course);
        await _db.SaveChangesAsync();
        return Results.NoContent();
    }
}
```

```csharp Features/Groups/EnrollStudent/EnrollStudentHandler.cs
using Lms.Api.Data;
using Lms.Api.Entities;
using Microsoft.EntityFrameworkCore;

namespace Lms.Api.Features.Groups.EnrollStudent;

public class EnrollStudentHandler
{
    private readonly DataContext _db;

    public EnrollStudentHandler(DataContext db)
    {
        _db = db;
    }

    public async Task<IResult> HandleAsync(Guid groupId, EnrollStudentRequest request)
    {
        var group = await _db.Groups
            .Include(g => g.Enrollments)
            .FirstOrDefaultAsync(g => g.Id == groupId);
        if (group is null)
            return Results.NotFound(GroupErrors.NotFound);

        if (!await _db.Students.AnyAsync(s => s.Id == request.StudentId))
            return Results.NotFound(StudentErrors.NotFound);

        var now = DateTime.UtcNow;
        var result = group.Enroll(request.StudentId, DateOnly.FromDateTime(now), now);
        if (result.IsFailure)
            return Results.Conflict(result.Error);

        try
        {
            await _db.SaveChangesAsync();
        }
        catch (DbUpdateConcurrencyException)
        {
            return Results.Conflict(GroupErrors.ConcurrentChange);
        }

        return Results.Ok(new EnrollmentResponse(group.Id, request.StudentId,
            group.EnrolledCount, group.Capacity));
    }
}
```
